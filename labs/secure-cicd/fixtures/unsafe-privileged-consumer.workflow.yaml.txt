name: Intentionally unsafe privileged artifact consumer

on:
  workflow_run:
    workflows:
      - Untrusted pull request validation
    types:
      - completed

permissions: write-all

jobs:
  release:
    runs-on: ubuntu-latest
    steps:
      - name: Download an artifact without binding producer, revision, or digest
        uses: actions/download-artifact@v4
        with:
          name: release
          path: release

      - name: Execute unverified artifact content with stored cloud credentials
        env:
          CLOUD_ACCESS_KEY: ${{ secrets.LONG_LIVED_CLOUD_ACCESS_KEY }}
          CLOUD_SECRET_KEY: ${{ secrets.LONG_LIVED_CLOUD_SECRET_KEY }}
        run: ./release/deploy.sh
