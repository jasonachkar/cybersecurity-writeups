15MAX_BYTES = 4096
16TIMEOUT = 2

19class Denied(Exception):
20    pass

23def public_address(value):
24    address = ipaddress.ip_address(value)
25    # Require ordinary global unicast. Reject mapped/transition forms too.
26    return (address.is_global and not address.is_multicast
27            and not address.is_reserved
28            and not getattr(address, "ipv4_mapped", None)
29            and not getattr(address, "sixtofour", None)
30            and not getattr(address, "teredo", None))

33def resolve(host, port):
34    return sorted({item[4][0] for item in socket.getaddrinfo(
35        host, port, type=socket.SOCK_STREAM)})

38class PinnedTLS(http.client.HTTPSConnection):
39    def __init__(self, host, ip, port):
40        super().__init__(host, port, timeout=TIMEOUT,
41                         context=ssl.create_default_context())
42        self.vetted_ip = ip
43 
44    def connect(self):
45        # Connect to the numeric result; keep original host for SNI/verification.
46        raw = socket.create_connection((self.vetted_ip, self.port), self.timeout)
47        try:
48            self.sock = self._context.wrap_socket(raw, server_hostname=self.host)
49        except Exception:
50            raw.close()
51            raise

54class FetchPolicy:
55    def __init__(self, allowed_hosts=(), lab_ip=None, lab_port=8080,
56                 resolver=resolve, connector=None):
57        self.allowed_hosts = frozenset(allowed_hosts)
58        self.lab_ip = lab_ip
59        self.lab_port = lab_port
60        self.resolver = resolver
61        self.connector = connector
62 
63    def destination(self, url):
64        if not isinstance(url, str) or not 1 <= len(url) <= 2048:
65            raise Denied("invalid_url")
66        if any(ord(c) < 33 or ord(c) == 127 for c in url) or "\\" in url:
67            raise Denied("ambiguous_url")
68        try:
69            parsed = urllib.parse.urlsplit(url)
70            host, port = parsed.hostname, parsed.port
71        except ValueError as error:
72            raise Denied("invalid_url") from error
73        if not host or parsed.username is not None or parsed.password is not None or parsed.fragment:
74            raise Denied("authority_or_fragment")
75        # One explicitly pinned synthetic HTTP origin replaces the public web.
76        lab_origin = (self.lab_ip is not None and host == "public.lab"
77                      and parsed.scheme == "http" and port == self.lab_port)
78        if not lab_origin and (parsed.scheme != "https" or port not in (None, 443)):
79            raise Denied("scheme_or_port")
80        if host not in self.allowed_hosts or host.endswith(".") or "%" in host:
81            raise Denied("host_not_allowed")
82        port = port or 443
83        try:
84            addresses = self.resolver(host, port)
85        except OSError as error:
86            raise Denied("dns_failed") from error
87        if not addresses:
88            raise Denied("dns_empty")
89        if lab_origin:
90            if set(addresses) != {self.lab_ip}:
91                raise Denied("lab_pin_mismatch")
92        elif not all(public_address(ip) for ip in addresses):
93            raise Denied("non_public_destination")
94        return parsed, addresses[0], port, lab_origin
95 
96    def fetch(self, url):
97        parsed, ip, port, lab_origin = self.destination(url)
98        if self.connector:
99            connection = self.connector(parsed.hostname, ip, port, lab_origin)
100        elif lab_origin:
101            connection = http.client.HTTPConnection(ip, port, timeout=TIMEOUT)
102        else:
103            connection = PinnedTLS(parsed.hostname, ip, port)
104        try:
105            path = urllib.parse.urlunsplit(("", "", parsed.path or "/", parsed.query, ""))
106            # No caller headers, cookies or credentials cross the fetch boundary.
107            connection.request("GET", path, headers={"Host": parsed.netloc,
108                                                     "Accept": "text/plain"})
109            response = connection.getresponse()
110            if 300 <= response.status < 400:
111                raise Denied("redirect_denied")
112            if response.status != 200:
113                raise Denied("upstream_status")
114            body = response.read(MAX_BYTES + 1)
115            if len(body) > MAX_BYTES:
116                raise Denied("response_too_large")
117            return body.decode("utf-8", errors="replace")
118        finally:
119            connection.close()

122def vulnerable_fetch(url):
123    """Intentionally lacks destination/redirect checks. Internal Docker net only."""
124    if not isinstance(url, str) or urllib.parse.urlsplit(url).scheme != "http":
125        raise Denied("http_lab_only")
126    # Do not inherit workstation proxy settings. Default redirect handling remains.
127    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}))
128    with opener.open(url, timeout=TIMEOUT) as response:
129        return response.read(MAX_BYTES).decode("utf-8", errors="replace")

132def handler(mode, policy=None, gateway=None, audit=None):
133    class Handler(BaseHTTPRequestHandler):
134        def log_message(self, *_args):
135            pass  # Request URLs/bodies must not reach default access logs.
136 
137        def do_POST(self):
138            decision, reason, status = "deny", "invalid_request", 400
139            result = {"error": reason}
140            started = time.monotonic()
141            try:
142                length = int(self.headers.get("Content-Length", "0"))
143                if self.path != "/fetch" or not 0 < length <= 4096:
144                    raise Denied("invalid_request")
145                value = json.loads(self.rfile.read(length))
146                if not isinstance(value, dict) or set(value) != {"url"}:
147                    raise Denied("invalid_request")
148                if mode == "vulnerable":
149                    body = vulnerable_fetch(value["url"])
150                elif mode == "secure":
151                    connection = http.client.HTTPConnection(gateway, timeout=3)
152                    try:
153                        connection.request("POST", "/fetch", json.dumps(value),
154                                           {"Content-Type": "application/json"})
155                        response = connection.getresponse()
156                        forwarded = json.loads(response.read(8192))
157                        if response.status != 200:
158                            raise Denied(forwarded.get("error", "gateway_denied"))
159                        body = forwarded["body"]
160                    finally:
161                        connection.close()
162                else:
163                    body = policy.fetch(value["url"])
164                status, decision, reason = 200, "allow", "fetched"
165                result = {"body": body}
166            except Denied as error:
167                reason = str(error)
168                status, result = 403, {"error": reason}
169            except (ValueError, TypeError, OSError, http.client.HTTPException):
170                reason = "invalid_or_unavailable"
171                status, result = 400, {"error": reason}
172            event = {"event": "tool_fetch", "mode": mode, "decision": decision,
173                     "reason": reason, "request_id": uuid.uuid4().hex,
174                     "duration_ms": round((time.monotonic() - started) * 1000)}
175            # No destination string, query, response, credential or user content.
176            if audit is not None:
177                audit.append(event)
178            else:
179                print(json.dumps(event), flush=True)
180            encoded = json.dumps(result).encode()
181            self.send_response(status)
182            self.send_header("Content-Type", "application/json")
183            self.send_header("Content-Length", str(len(encoded)))
184            self.end_headers()
185            self.wfile.write(encoded)
186    return Handler

189if __name__ == "__main__":
190    mode = os.environ.get("MODE", "secure")
