2"""Detection engine for labs/linux-host-telemetry: events in, deduplicated alerts out.
3 
4Rules are plain SQL files in rules/ run against an in-memory SQLite table, so every
5detection is a readable query with no hidden logic. The engine adds only what SQL
6should not carry: schema validation, allowlist evaluation (owner + expiry), source
7coverage ("blind" instead of "clean"), batching clocks, and deduplication.
8 
9Usage: engine.py --events events.jsonl [--scenarios scenarios.json] [--sources exec,file,net]
10"""

39def validate(event, line_no):
40    for field in REQUIRED:
41        if field not in event:
42            raise ValidationError(f"line {line_no}: missing required field {field!r}")
43    if event["event"] not in EVENT_TYPES:
44        raise ValidationError(f"line {line_no}: unknown event type {event['event']!r}")
45    if EVENT_TYPES[event["event"]] != event["source"]:
46        raise ValidationError(f"line {line_no}: event {event['event']!r} cannot come from source {event['source']!r}")
47    if event["origin"] not in ORIGINS:
48        raise ValidationError(f"line {line_no}: unknown origin {event['origin']!r}")
49    if event["synthetic"] is not ORIGINS[event["origin"]]:
50        raise ValidationError(f"line {line_no}: synthetic={event['synthetic']!r} contradicts origin {event['origin']!r}")
51    for field in ("ts", "ingest_ts"):
52        parse_ts(event[field])
53    if parse_ts(event["ingest_ts"]) < parse_ts(event["ts"]):
54        raise ValidationError(f"line {line_no}: ingest_ts precedes ts")
55    if event["event"] == "process_start":
56        for field in ("pid", "pstart", "ppid", "ppstart", "exe"):
57            if event.get(field) is None:
58                raise ValidationError(f"line {line_no}: process_start without {field}")

73class Allowlist:
74    def __init__(self, path=HERE / "allowlist.json"):
75        self.entries = json.loads(pathlib.Path(path).read_text())["entries"] if path else []
76        for entry in self.entries:
77            for key in ("rule", "pattern", "owner", "reason", "expires"):
78                if not entry.get(key):
79                    raise ValidationError(f"allowlist entry {entry} lacks {key}: exceptions need an owner, a reason and an expiry")
80            re.compile(entry["pattern"])
81 
82    def allowed(self, rule, ts, exe, cmdline, path, parent_exe):
83        day = ts[:10]
84        for entry in self.entries:
85            if entry["rule"] != rule or day > entry["expires"]:
86                continue
87            value = {"exe": exe, "cmdline": cmdline, "path": path}.get(entry.get("field", "exe"), "") or ""
88            if not re.search(entry["pattern"], value):
89                continue
90            if entry.get("parent_exe") and entry["parent_exe"] != (parent_exe or ""):
91                continue
92            if entry.get("cmdline_pattern") and not re.search(entry["cmdline_pattern"], cmdline or ""):
93                continue
94            return True
95        return False

98def parse_rule(path):
99    text = path.read_text()
100    meta = dict(re.findall(r"^-- (\w+): (.*)$", text, re.M))
101    for key in ("id", "title", "severity", "requires", "response"):
102        if key not in meta:
103            raise ValidationError(f"{path.name}: missing '-- {key}:' header")
104    meta["requires"] = {s.strip() for s in meta["requires"].split(",")}
105    # Optional: event fields the rule cannot work without (a source can be present but lack a field).
106    meta["requires_fields"] = {s.strip() for s in meta.get("requires_fields", "").split(",") if s.strip()}
107    meta["sql"] = text
108    return meta

115def connect(events, allowlist):
116    db = sqlite3.connect(":memory:")
117    db.row_factory = sqlite3.Row
118    db.create_function("basename", 1, lambda p: (p or "").rsplit("/", 1)[-1].split(" (deleted)")[0])
119    db.create_function("regexp", 2, lambda pattern, value: 1 if value is not None and re.search(pattern, value) else 0)
120    db.create_function("is_allowed", 6, lambda rule, ts, exe, cmd, path, parent: 1 if allowlist.allowed(rule, ts, exe, cmd, path, parent) else 0)
121    db.execute(f"CREATE TABLE events ({', '.join(c + (' INTEGER PRIMARY KEY' if c == 'id' else '') for c in COLUMNS)})")
122    db.execute("CREATE INDEX proc_identity ON events (host, boot_id, pid, pstart)")
123    rows = [[(1 if e.get(c) is True else 0 if e.get(c) is False else e.get(c)) for c in COLUMNS] for e in events]
124    db.executemany(f"INSERT INTO events VALUES ({','.join('?' * len(COLUMNS))})", rows)
125    return db

128def run(events, rules, allowlist, sources=None):
129    """Return (alerts, coverage). sources=None means: whatever the events contain."""
130    available = set(sources) if sources is not None else {e["source"] for e in events}
131    usable = [e for e in events if e["source"] in available]
132    db = connect(usable, allowlist)
133    by_id = {e["id"]: e for e in usable}
134    coverage, hits = {}, []
135    for rule in rules:
136        missing = sorted(rule["requires"] - available)
137        missing_fields = sorted(f for f in rule["requires_fields"] if not any(e.get(f) is not None for e in usable))
138        if missing or missing_fields:
139            coverage[rule["id"]] = {"status": "blind", "missing_sources": missing, "missing_fields": missing_fields}
140            continue
141        coverage[rule["id"]] = {"status": "evaluated", "missing_sources": [], "missing_fields": []}
142        for row in db.execute(rule["sql"]):
143            hits.append({"rule": rule["id"], "severity": rule["severity"], "event_id": row["event_id"], "host": row["host"],
144                         "subject": row["subject"], "detail": row["detail"]})
145    # Deduplicate: same (rule, host, subject) within the window collapses into one alert with a count.
146    hits.sort(key=lambda h: by_id[h["event_id"]]["ingest_ts"])
147    alerts, open_alert = [], {}
148    for hit in hits:
149        event = by_id[hit["event_id"]]
150        key = (hit["rule"], hit["host"], hit["subject"])
151        ingest = parse_ts(event["ingest_ts"])
152        current = open_alert.get(key)
153        if current and (ingest - current["_last"]).total_seconds() <= DEDUP_WINDOW_SECONDS:
154            current["count"] += 1
155            current["_last"] = ingest
156            current["event_ids"].append(hit["event_id"])
157            # Evaluation-only: which labeled scenarios were folded into this alert. A real
158            # pipeline has no such field - the folded activity is visible only as the count.
159            current["_scenarios"].add(event.get("scenario"))
160            continue
161        batch_end = dt.datetime.fromtimestamp(((ingest.timestamp() // BATCH_SECONDS) + 1) * BATCH_SECONDS, dt.timezone.utc)
162        alert = dict(hit, scenario=event.get("scenario"), count=1, event_ids=[hit["event_id"]], _last=ingest,
163                     _scenarios={event.get("scenario")},
164                     event_ts=event["ts"], ingest_ts=event["ingest_ts"],
165                     alert_ts=batch_end.strftime("%Y-%m-%dT%H:%M:%S.000Z"),
166                     synthetic_latency_seconds=round((batch_end - parse_ts(event["ts"])).total_seconds(), 1))
167        open_alert[key] = alert
168        alerts.append(alert)
169    for alert in alerts:
170        alert.pop("_last")
171        alert["scenarios"] = sorted(alert.pop("_scenarios"))
172    return alerts, coverage

175def score(alerts, scenarios, rules, coverage):
176    """Scenario-level confusion matrix per rule; background noise is counted separately.
177 
178    Class semantics (see generator.py):
179      malicious_pattern   rule in expected_rules -> TP if it fired else FN
180      evasion             rule in evades         -> FN if it stayed silent else TP (the known miss closed)
181      benign_lookalike    any rule               -> FP if it fired else TN (expected_rules lists KNOWN residual FPs)
182      control             any rule               -> FP if it fired else TN
183    A rule firing on a malicious scenario it is not responsible for is "incidental" (neither TP nor FP).
184    unexpected_outcomes compares the actual result with the DOCUMENTED one, so a documented false
185    positive or miss that silently changes (better or worse) is surfaced.
186    """
187    fired = {}
188    for alert in alerts:
189        for scenario in alert["scenarios"]:  # includes scenarios folded in by deduplication
190            fired.setdefault(scenario, set()).add(alert["rule"])
191    table = {r["id"]: {"TP": 0, "FP": 0, "FN": 0, "TN": 0, "incidental": 0, "blind": coverage[r["id"]]["status"] == "blind"} for r in rules}
192    unexpected = []
193    for sc in scenarios:
194        for rule in rules:
195            rid = rule["id"]
196            if table[rid]["blind"]:
197                continue
198            did = rid in fired.get(sc["id"], set())
199            documented = rid in sc["expected_rules"]
200            if sc["class"] == "malicious_pattern":
201                if documented:
202                    table[rid]["TP" if did else "FN"] += 1
203                elif did:
204                    table[rid]["incidental"] += 1
205            elif sc["class"] == "evasion":
206                if rid in sc.get("evades", []):
207                    table[rid]["TP" if did else "FN"] += 1
208                elif did:
209                    table[rid]["FP"] += 1
210                else:
211                    table[rid]["TN"] += 1
212            else:  # benign_lookalike, control
213                table[rid]["FP" if did else "TN"] += 1
214            if sc["class"] == "evasion" and rid in sc.get("evades", []):
215                documented = False
216            if did != documented and not (sc["class"] == "malicious_pattern" and not documented):
217                unexpected.append({"scenario": sc["id"], "rule": rid, "documented_to_fire": documented, "fired": did, "class": sc["class"]})
218    background = [a for a in alerts if "bg" in a["scenarios"]]
219    return table, unexpected, background

2"""Deterministic SYNTHETIC host telemetry with labeled scenarios.
3 
4Nothing here records or simulates a real intrusion. Every event is generated from a
5fixed seed, carries "synthetic": true, and describes a behaviour only as data
6(a path, an exe name, a destination in a documentation address range). No malicious
7code exists in this lab. Scenarios are labeled so detections can be scored:
8 
9  malicious_pattern  - a behaviour a detection SHOULD flag
10  benign_lookalike   - legitimate activity that resembles a malicious pattern
11  evasion            - a malicious pattern the shipped rules are KNOWN to miss
12  control            - ordinary activity that nothing should flag
13 
14Usage: generator.py [--seed N] [--out events.jsonl] [--scenarios scenarios.json]
15"""

30class Builder:
31    def __init__(self, seed):
32        self.rng = random.Random(seed)
33        self.events = []
34        self.scenarios = []
35        self.next_pid = 2000
36        self.clock = 0.0  # seconds since T0
37        self.start_ns = 1_000_000_000
38 
39    # ---- primitives -------------------------------------------------------
40    def tick(self, seconds=None):
41        self.clock += seconds if seconds is not None else self.rng.uniform(0.2, 3.0)
42        return self.clock
43 
44    def ts(self, offset):
45        return (T0 + dt.timedelta(seconds=offset)).strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"
46 
47    def proc(self, ppid, ppstart, exe, cmdline, uid=0, tty=None, pid=None):
48        pid = pid or self.next_pid
49        if pid == self.next_pid:
50            self.next_pid += self.rng.randint(1, 9)
51        self.start_ns += self.rng.randint(10_000_000, 900_000_000)
52        return {"pid": pid, "pstart": self.start_ns, "ppid": ppid, "ppstart": ppstart, "exe": exe, "cmdline": cmdline, "uid": uid, "tty": tty}
53 
54    def emit(self, scenario, kind, source, proc, lag=None, **extra):
55        t = self.tick()
56        lag = self.rng.uniform(0.2, 2.5) if lag is None else lag
57        event = {
58            "ts": self.ts(t), "ingest_ts": self.ts(t + lag), "host": HOST, "boot_id": BOOT,
59            "source": source, "event": kind, "pid": proc["pid"], "pstart": proc["pstart"],
60            "ppid": proc["ppid"], "ppstart": proc["ppstart"], "uid": proc["uid"], "exe": proc["exe"],
61            "cmdline": proc["cmdline"], "tty": proc["tty"], "scenario": scenario, "synthetic": True, "origin": "synthetic-generator",
62        }
63        event.update(extra)
64        self.events.append(event)
65        return event
66 
67    def start(self, scenario, proc, lag=None):
68        return self.emit(scenario, "process_start", "exec", proc, lag=lag)
69 
70    def scenario(self, sid, name, klass, expect, note, requires=(), evades=()):
71        self.scenarios.append({"id": sid, "name": name, "class": klass, "expected_rules": sorted(expect), "evades": sorted(evades), "note": note, "requires_sources": sorted(requires)})
72        return sid

111            b.start("bg", p)

114            p = b.proc(cron["pid"], cron["pstart"], "/usr/sbin/logrotate", "logrotate /etc/logrotate.conf")

117        else:
118            p = b.proc(cron["pid"], cron["pstart"], "/usr/bin/dpkg", "dpkg --configure -a")

120            b.emit("bg", "file_write", "file", p, path=R.choice(["/etc/cron.d/logrotate", "/etc/default/locale", "/etc/passwd"]))

131    b.start(s, helper)
132    bash = b.proc(helper["pid"], helper["pstart"], "/usr/bin/bash", "bash -i", uid=33)
133    b.start(s, bash)
134 
135    s = b.scenario("S03", "php-fpm runs an allowlisted image converter via sh", "benign_lookalike", [], "suppressed by an allowlist entry with an owner and an expiry")
136    b.start(s, b.proc(phpfpm_w["pid"], phpfpm_w["pstart"], "/bin/sh", "sh -c convert /tmp/in.png -resize 200x /tmp/out.png", uid=33))
137 
138    s = b.scenario("S04", "gunicorn app shells out to git", "benign_lookalike", ["LNX-001"], "RESIDUAL FALSE POSITIVE: legitimate, not allowlisted; needs an owner decision")
139    b.start(s, b.proc(gunicorn["pid"], gunicorn["pstart"], "/bin/sh", "sh -c git rev-parse HEAD", uid=1001))
140 
141    s = b.scenario("S05", "admin reloads nginx from an ssh session", "control", [], "parent is the admin shell, not the service")
142    b.start(s, b.proc(admin_bash["pid"], admin_bash["pstart"], "/usr/sbin/nginx", "nginx -t", uid=1000, tty="pts/0"))

144    # S06: PID reuse. Process A (pid 61000) was a child of nginx; it exits. The kernel reuses
145    # 61000 for B, a child of the admin shell, which then starts a shell C. Resolving C's parent by pid
146    # alone can land on A (a nginx descendant) and wrongly alert; resolving by pid + start time cannot.

32def classify(ip):

57    def emit(ts, kind, source, proc, **extra):
58        event = {"id": len(events) + 1, "ts": iso(ts), "ingest_ts": iso(ts), "host": host, "boot_id": boot, "source": source,
59                 "event": kind, "pid": proc["pid"], "pstart": proc["pstart"], "ppid": proc["ppid"], "ppstart": proc["ppstart"],
60                 "uid": 0, "exe": proc["exe"], "cmdline": proc["cmdline"], "tty": None, "scenario": "capture",
61                 "synthetic": False, "origin": "strace-capture"}
62        event.update(extra)
63        events.append(event)

65    for raw in lines:
66        m = LINE.match(raw.rstrip("\n"))
67        if not m:
68            continue
69        pid, ts, rest = int(m.group(1)), m.group(2), m.group(3)
70        if rest.startswith("+++") or rest.startswith("---"):
71            continue
72        name = args = ret = None
73        c = CALL.match(rest)
74        if c:
75            name, args, ret = c.group(1), c.group(2), c.group(3)
76        else:
77            r = RESUMED.match(rest)
78            if r:
79                name, args, ret = r.group(1), r.group(2), r.group(3)
80        if name is None or ret in (None, "?"):
81            continue
82        ret = int(ret)
83        if pid not in procs:
84            # The first traced process was already running when the tracer attached.
85            new_proc(pid, 0, root_exe, root_exe)
86        proc = procs[pid]
87        if name in ("clone", "fork", "vfork", "clone3"):
88            if ret > 0:
89                new_proc(ret, pid, proc["exe"], proc["cmdline"])
90            continue
91        if ret < 0:
92            continue  # failed syscalls are not behaviour
93        quoted = QUOTED.findall(args)
94        if name == "execve" and quoted:
95            argv = QUOTED.findall(args.split("], ")[0]) if "[" in args else quoted
96            proc["exe"], proc["cmdline"] = quoted[0], " ".join(argv[1:]) if len(argv) > 1 else quoted[0]
97            emit(ts, "process_start", "exec", proc)
98        elif name in ("open", "openat") and quoted:
99            path, flags = quoted[0], args
100            writes = any(f in flags for f in ("O_WRONLY", "O_RDWR", "O_CREAT", "O_TRUNC", "O_APPEND"))
101            if writes:
102                emit(ts, "file_write", "file", proc, path=path)
103                if "O_TRUNC" in flags:
104                    emit(ts, "file_truncate", "file", proc, path=path)
105            elif not path.startswith(SKIP_READ_PREFIXES):
106                emit(ts, "file_read", "file", proc, path=path)
107        elif name in ("chmod", "fchmodat") and quoted:
108            mode = re.search(r",\s*(0[0-7]+)\s*(?:\)|,|$)", args)
109            if mode:
110                emit(ts, "file_chmod", "file", proc, path=quoted[0], mode=int(mode.group(1), 8), is_dir=False)
111        elif name == "truncate" and quoted:
112            emit(ts, "file_truncate", "file", proc, path=quoted[0])
113        elif name in ("unlink", "unlinkat") and quoted:
114            emit(ts, "file_delete", "file", proc, path=quoted[0])
115        elif name == "connect":
116            ip = re.search(r'inet_addr\("([^"]+)"\)', args)
117            port = re.search(r"sin_port=htons\((\d+)\)", args)
118            if ip and port:

    What was not tested¶

    • Any production sensor: auditd, eBPF, Falco, Tetragon, osquery. The schema is inspired by what they emit; none was run.
    • Detection on real fleet data, real adversary tradecraft, or adversaries who know the rules. Evasion is represented by two hand-written cases.
    • strace as a sensor: its overhead and blind spots were not measured; it is a capture convenience.
    • Rules LNX-001, 003, 004 and 008 against real kernel data (the capture exercises 002, 005, 006, 007, 009 and one negative for 004).
    • Time to detect and time to contain: the lab reports synthetic latency from synthetic clocks only.
    • Response actions: they are written guidance, never executed.

    Production considerations¶

    See the article for rollout, detection, operability, false positives, blast radius and rollback. In short: ship rules to a few hosts in report-only mode, own every exception (owner + expiry), page on nothing until the residual false-positive sources for your environment are known, and treat a blind rule as an incident in the telemetry pipeline, not as a quiet host.
